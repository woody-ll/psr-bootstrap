# 1. Меняем порт RDP с 3389 на 64121
Set-ItemProperty 'HKLM:\System\CurrentControlSet\Control\Terminal Server\WinStations\RDP-Tcp' -Name PortNumber -Value 64121

# 2. Старые правила firewall на 64121 (если есть) убираем
Get-NetFirewallRule -DisplayName "RDP Custom Port" -ErrorAction SilentlyContinue | Remove-NetFirewallRule

# 3. Открываем новое firewall правило на 64121
New-NetFirewallRule -DisplayName "RDP Custom Port" -Direction Inbound -Protocol TCP -LocalPort 64121 -Action Allow -Profile Domain,Private,Public

# 4. Рестартуем сервис чтобы порт применился
Restart-Service TermService -Force

# 5. Проверка
Get-NetTCPConnection -State Listen | Where-Object LocalPort -eq 64121
